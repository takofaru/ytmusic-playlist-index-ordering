// ==UserScript==
// @name         YouTube Music Inline Index Reorder
// @namespace    http://tampermonkey.net/
// @version      2.5.2
// @description  Injects an index input box onto every song row to move tracks by typing a position number. Only active when the playlist is in Manual ordering mode.
// @author       You
// @match        https://music.youtube.com/*
// @icon         https://www.google.com/s2/favicons?sz=64&domain=youtube.com
// @grant        none
// @license      MIT
// ==/UserScript==

(function() {
    'use strict';

    // Inject CSS - z-index set to 1 so popup menus overlay correctly
    const style = document.createElement('style');
    style.textContent = `
        .ytm-custom-index-box {
            width: 40px !important;
            min-width: 40px !important;
            margin-left: 8px !important;
            margin-right: 12px !important;
            padding: 3px 4px !important;
            background: rgba(0, 0, 0, 0.6) !important;
            color: #ffffff !important;
            border: 1px solid rgba(255, 255, 255, 0.4) !important;
            border-radius: 4px !important;
            text-align: center !important;
            font-size: 13px !important;
            font-weight: bold !important;
            z-index: 1 !important;
            pointer-events: auto !important;
            flex-shrink: 0 !important;
            align-self: center !important;
        }
        .ytm-custom-index-box:focus {
            border-color: #ff0000 !important;
            background: rgba(0, 0, 0, 0.95) !important;
            outline: none !important;
        }
    `;
    document.head.appendChild(style);

    // Recursively look for the sort option list (subMenuItems containing
    // ACTION_SET_PLAYLIST_VIDEO_ORDER). Skips big track lists for speed.
    function findSortItems(obj, depth) {
        if (!obj || typeof obj !== 'object' || depth > 8 || obj instanceof Node) return null;
        if (Array.isArray(obj.subMenuItems) && obj.subMenuItems.some(i =>
            i?.serviceEndpoint?.playlistEditEndpoint?.actions?.[0]?.action === 'ACTION_SET_PLAYLIST_VIDEO_ORDER')) {
            return obj.subMenuItems;
        }
        for (const k of Object.keys(obj)) {
            if (k === 'contents' || k === 'continuations' || k === 'trackingParams' || k === 'clickTrackingParams') continue;
            let v;
            try { v = obj[k]; } catch (e) { continue; }
            if (v && typeof v === 'object') {
                const r = findSortItems(v, depth + 1);
                if (r) return r;
            }
        }
        return null;
    }

    function getSortItems() {
        const els = document.querySelectorAll('*');
        for (const el of els) {
            if (!el.tagName.startsWith('YTMUSIC-') || el.tagName === 'YTMUSIC-RESPONSIVE-LIST-ITEM-RENDERER') continue;
            const found = findSortItems(el.data, 0) || findSortItems(el.__data?.data, 0);
            if (found) return found;
        }
        return null;
    }

    // Read the selected option straight from the rendered sort dropdown:
    // <tp-yt-paper-listbox> > <a class="yt-dropdown-menu iron-selected" aria-selected="true">
    // Returns true/false (manual or not), or null if the dropdown can't be found.
    function manualFromDropdown() {
        let listboxes = Array.from(document.querySelectorAll('ytmusic-sort-filter-sub-menu-renderer tp-yt-paper-listbox'));
        if (listboxes.length === 0) {
            // Fallback: any listbox whose options look like sort options
            listboxes = Array.from(document.querySelectorAll('tp-yt-paper-listbox')).filter(lb =>
                /manual/i.test(lb.textContent) && /newest|oldest|title|artist|album/i.test(lb.textContent));
        }

        for (const lb of listboxes) {
            const anchors = Array.from(lb.querySelectorAll('a.yt-dropdown-menu'));
            if (anchors.length < 2) continue;

            const selIdx = anchors.findIndex(a => a.classList.contains('iron-selected') || a.getAttribute('aria-selected') === 'true');
            if (selIdx === -1) continue;
            const sel = anchors[selIdx];

            // 1) Language-independent: option data (playlistVideoOrder 0 = Manual)
            const d = sel.data || sel.__data?.data || sel.__dataHost?.data;
            const order = d?.serviceEndpoint?.playlistEditEndpoint?.actions?.[0]?.playlistVideoOrder;
            if (typeof order === 'number') return order === 0;

            // 2) Position of Manual among the options, from the menu data
            const items = getSortItems();
            if (items && items.length === anchors.length) {
                const manualIdx = items.findIndex(i => i.serviceEndpoint?.playlistEditEndpoint?.actions?.[0]?.playlistVideoOrder === 0);
                if (manualIdx !== -1) return selIdx === manualIdx;
            }

            // 3) Text match
            if (/manual/i.test(sel.textContent)) return true;
            if (/newest|oldest|top voted|title|artist|album/i.test(sel.textContent)) return false;
        }
        return null;
    }

    let lastLoggedSort = null;

    // Returns true only when the playlist is in Manual ordering mode.
    // Selected option's playlistVideoOrder: 0 = Manual (language-independent).
    function isManualSortActive() {
        const fromDom = manualFromDropdown();
        if (fromDom !== null) {
            const key = 'dom' + fromDom;
            if (lastLoggedSort !== key) {
                console.log('[Reorder] Sort mode from dropdown: manual =', fromDom);
                lastLoggedSort = key;
            }
            return fromDom;
        }

        const items = getSortItems();
        if (!items) {
            if (lastLoggedSort !== 'none') console.warn('[Reorder] Sort menu data not found; assuming Manual.');
            lastLoggedSort = 'none';
            return true;
        }

        const selected = items.find(i => i.selected);
        if (!selected) return true;

        const order = selected.serviceEndpoint?.playlistEditEndpoint?.actions?.[0]?.playlistVideoOrder;
        const manual = typeof order === 'number'
            ? order === 0
            : /manual/i.test(String(selected.title?.runs?.[0]?.text || selected.title || ''));

        if (lastLoggedSort !== String(order)) {
            console.log('[Reorder] Sort mode:', selected.title, '| order =', order, '| manual =', manual);
            lastLoggedSort = String(order);
        }
        return manual;
    }

    // Remove every injected input (used when not in Manual mode)
    function removeIndexes() {
        document.querySelectorAll('.ytm-custom-index-box').forEach(el => el.remove());
    }

    // Generate YouTube's required SAPISIDHASH header
    async function getAuthorizationHeader() {
        const getCookie = (name) => document.cookie.match(new RegExp('(?:^|; )' + name + '=([^;]*)'))?.[1];
        const sapisid = getCookie('SAPISID') || getCookie('__Secure-3PAPISID');

        if (!sapisid) {
            console.error('[Reorder] SAPISID cookie not found. Log into YouTube Music.');
            return null;
        }

        const origin = window.location.origin;
        const time = Math.floor(Date.now() / 1000);
        const data = `${time} ${sapisid} ${origin}`;

        const encoder = new TextEncoder();
        const hashBuffer = await crypto.subtle.digest('SHA-1', encoder.encode(data));
        const hashArray = Array.from(new Uint8Array(hashBuffer));
        const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

        return `SAPISIDHASH ${time}_${hashHex}`;
    }

    // Extract playlistSetVideoId from Polymer DOM node properties
    function getSetVideoId(row) {
        if (!row) return null;

        let setVideoId = row.data?.playlistItemData?.playlistSetVideoId ||
                         row.__data?.data?.playlistItemData?.playlistSetVideoId ||
                         row.querySelector('[data-set-video-id]')?.getAttribute('data-set-video-id');
        if (setVideoId) return setVideoId;

        const data = row.data || row.__data?.data || row.dataData;
        if (data) {
            try {
                const str = JSON.stringify(data);
                const match = str.match(/"setVideoId"\s*:\s*"([^"]+)"/);
                if (match && match[1]) return match[1];
            } catch (e) {}
        }

        return null;
    }

    // Send API Request and update DOM
    async function moveTrack(playlistId, sourceRow, targetRow, isMovingUp) {
        const cleanPlaylistId = playlistId.replace(/^VL/, '');
        const movedSetVideoId = getSetVideoId(sourceRow);
        const targetSetVideoId = getSetVideoId(targetRow);

        if (!movedSetVideoId || !targetSetVideoId) {
            console.error('[Reorder Error] Missing playlistSetVideoId for source or target.');
            alert("Could not retrieve track ID. Ensure you own this playlist.");
            return false;
        }

        // InnerTube expects movedSetVideoIdSuccessor for BEFORE and movedSetVideoIdPredecessor for AFTER
        const action = isMovingUp
            ? {
                action: 'ACTION_MOVE_VIDEO_BEFORE',
                setVideoId: movedSetVideoId,
                movedSetVideoIdSuccessor: targetSetVideoId
              }
            : {
                action: 'ACTION_MOVE_VIDEO_AFTER',
                setVideoId: movedSetVideoId,
                movedSetVideoIdPredecessor: targetSetVideoId
              };

        const context = window.ytcfg?.get('INNERTUBE_CONTEXT');
        const apiKey = window.ytcfg?.get('INNERTUBE_API_KEY');
        const authHeader = await getAuthorizationHeader();

        if (!context || !apiKey || !authHeader) {
            console.error('[Reorder Error] Missing API configuration or authentication headers.');
            return false;
        }

        const headers = {
            'Content-Type': 'application/json',
            'Authorization': authHeader,
            'X-Origin': window.location.origin,
            'X-YouTube-Client-Name': window.ytcfg.get('INNERTUBE_CONTEXT_CLIENT_NAME') || '67',
            'X-YouTube-Client-Version': window.ytcfg.get('INNERTUBE_CONTEXT_CLIENT_VERSION') || '1.20261104.00.00'
        };

        const authUser = window.ytcfg.get('SESSION_INDEX');
        if (authUser !== undefined) headers['X-Goog-AuthUser'] = authUser;

        const pageId = window.ytcfg.get('DELEGATED_SESSION_ID');
        if (pageId) headers['X-Goog-PageId'] = pageId;

        try {
            const response = await fetch(`https://music.youtube.com/youtubei/v1/browse/edit_playlist?key=${apiKey}`, {
                method: 'POST',
                headers: headers,
                credentials: 'include',
                body: JSON.stringify({
                    context: context,
                    playlistId: cleanPlaylistId,
                    actions: [action]
                })
            });

            const data = await response.json();
            console.log('[Reorder Response]', data);

            if (data.status === 'STATUS_SUCCEEDED' || (data.actions && !data.error)) {
                const container = sourceRow.parentElement;
                if (isMovingUp) {
                    container.insertBefore(sourceRow, targetRow);
                } else {
                    container.insertBefore(sourceRow, targetRow.nextSibling);
                }
                return true;
            } else {
                console.error('[Reorder Error] Edit action rejected by YouTube:', data);
                alert('Playlist update failed on YouTube server.');
                return false;
            }
        } catch (err) {
            console.error('[Reorder Network Error]', err);
            return false;
        }
    }

    function getPlaylistId() {
        return new URLSearchParams(window.location.search).get('list');
    }

    // Attach input elements and sync indices
    function updateIndexes() {
        const playlistId = getPlaylistId();
        if (!playlistId) { removeIndexes(); return; }

        // Only operate in Manual ordering mode
        if (!isManualSortActive()) { removeIndexes(); return; }

        const container = document.querySelector('ytmusic-playlist-shelf-renderer #contents') ||
                          document.querySelector('ytmusic-section-list-renderer #contents');
        if (!container) return;

        const rows = Array.from(container.querySelectorAll('ytmusic-responsive-list-item-renderer'));
        if (rows.length === 0) return;

        rows.forEach((row, index) => {
            let input = row.querySelector('.ytm-custom-index-box');

            if (!input) {
                input = document.createElement('input');
                input.type = 'number';
                input.className = 'ytm-custom-index-box';
                input.setAttribute('draggable', 'false');

                // Stop YouTube Music keyboard/mouse listener hijacking
                ['click', 'mousedown', 'mouseup', 'pointerdown', 'pointerup', 'keydown', 'keyup', 'keypress', 'dragstart'].forEach(evt => {
                    input.addEventListener(evt, (e) => e.stopPropagation());
                });

                // Handle Submit on Enter key press
                input.addEventListener('keydown', async (e) => {
                    if (e.key === 'Enter') {
                        e.preventDefault();
                        e.stopPropagation();

                        if (!isManualSortActive()) { removeIndexes(); return; }

                        const targetPos = parseInt(input.value, 10);
                        if (isNaN(targetPos)) return;

                        const currentRows = Array.from(container.querySelectorAll('ytmusic-responsive-list-item-renderer'));
                        const currentIndex = currentRows.indexOf(row);
                        const targetIndex = targetPos - 1;

                        if (currentIndex === -1 || targetIndex === currentIndex || targetIndex < 0 || targetIndex >= currentRows.length) return;

                        const targetRow = currentRows[targetIndex];
                        const isMovingUp = targetIndex < currentIndex;

                        input.disabled = true;
                        input.style.opacity = '0.4';

                        const success = await moveTrack(playlistId, row, targetRow, isMovingUp);

                        input.disabled = false;
                        input.style.opacity = '1';

                        if (success) {
                            updateIndexes();
                        }
                    }
                });

                // Attach to far-left of row flex container
                row.prepend(input);
            }

            input.title = "Type position index and press Enter";

            // Sync numerical input value when user is not actively typing
            if (document.activeElement !== input) {
                input.value = index + 1;
            }
        });
    }

    setInterval(updateIndexes, 1000);
})();
